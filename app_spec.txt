<project_specification>
  <project_name>Courthouse Digital Signage System</project_name>

  <overview>
    A digital signage system for the Frank E. Moss U.S. Courthouse that displays daily court docket information, announcements, and real-time data (weather, time) on HDMI-connected displays positioned outside courtrooms. The system consists of a full-screen HTML5 display client for kiosk screens, a React-based admin portal for content management, and a Node.js REST API with real-time WebSocket updates.
  </overview>

  <technology_stack>
    <frontend_display>
      <framework>HTML5 + CSS3 + Vanilla JavaScript</framework>
      <styling>Custom CSS with CSS Variables for theming</styling>
      <target>Chromium-based kiosk mode browsers</target>
      <resolution>1920x1080 (Full HD) primary</resolution>
    </frontend_display>
    <frontend_admin>
      <framework>React 18 with TypeScript</framework>
      <styling>Tailwind CSS</styling>
      <state_management>React Query (TanStack Query) for server state</state_management>
      <routing>React Router v6</routing>
      <forms>React Hook Form with Zod validation</forms>
      <real_time>Socket.io-client</real_time>
    </frontend_admin>
    <backend>
      <runtime>Node.js 20+ with Express</runtime>
      <language>TypeScript</language>
      <database>SQLite (development) / PostgreSQL (production) via Prisma ORM</database>
      <real_time>Socket.io</real_time>
      <validation>Zod</validation>
      <authentication>JWT tokens with role-based access</authentication>
    </backend>
    <communication>
      <api>RESTful API with JSON</api>
      <real_time>Socket.io WebSocket events</real_time>
      <refresh_interval>30-60 seconds for display polling</refresh_interval>
    </communication>
  </technology_stack>

    <backend_api>
      <authentication_endpoints>
        - POST /api/auth/login - User login with JWT response
        - POST /api/auth/logout - Logout and invalidate token
        - POST /api/auth/refresh - Refresh JWT token
        - GET /api/auth/me - Get current user info
      </authentication_endpoints>
      <docket_endpoints>
        - GET /api/docket - List entries with filters (date, courtroom, status)
        - POST /api/docket - Create single entry
        - POST /api/docket/bulk - Bulk create entries
        - GET /api/docket/:id - Get single entry
        - PUT /api/docket/:id - Update entry
        - DELETE /api/docket/:id - Delete entry
        - DELETE /api/docket/clear - Clear by date/courtroom with archive option
      </docket_endpoints>
      <display_endpoints>
        - GET /api/displays - List all displays
        - POST /api/displays - Register new display (returns API key)
        - GET /api/displays/:id - Get display details
        - PUT /api/displays/:id - Update display configuration
        - DELETE /api/displays/:id - Remove display
        - GET /api/displays/:id/config - Get display config (for display client)
        - GET /api/displays/:id/docket - Get filtered docket for display
        - POST /api/displays/:id/heartbeat - Display heartbeat
      </display_endpoints>
      <announcement_endpoints>
        - GET /api/announcements - List all (with ?active=true filter)
        - POST /api/announcements - Create announcement
        - GET /api/announcements/:id - Get single announcement
        - PUT /api/announcements/:id - Update announcement
        - DELETE /api/announcements/:id - Delete announcement
      </announcement_endpoints>
      <user_endpoints>
        - GET /api/users - List all users (admin only)
        - POST /api/users - Create user (admin only)
        - GET /api/users/:id - Get user details
        - PUT /api/users/:id - Update user
        - DELETE /api/users/:id - Deactivate user
      </user_endpoints>
      <api_key_endpoints>
        - GET /api/api-keys - List API keys (admin only)
        - POST /api/api-keys - Create API key
        - DELETE /api/api-keys/:id - Revoke API key
      </api_key_endpoints>
      <system_endpoints>
        - GET /api/health - Health check with DB status
        - GET /api/settings - Get global settings
        - PUT /api/settings - Update global settings (admin only)
        - GET /api/audit-logs - Query audit logs (admin only)
      </system_endpoints>
      <websocket_events>
        - docket:update - Full docket refresh
        - docket:entry:update - Single entry changed
        - announcement:new - New announcement added
        - announcement:remove - Announcement removed
        - display:refresh - Force display page refresh
        - display:message - One-time message overlay
      </websocket_events>
    </backend_api>

        - id (string, primary key, e.g., "display-321-main")
        - name
        - location
        - judge_filter (optional)
        - courtroom_filter (optional)
        - chapter_filter (array, optional)
        - show_stricken (boolean)
        - show_zoom_info (boolean)
        - highlight_current (boolean)
        - theme (string)
        - columns (array)
        - show_weather (boolean)
        - weather_location (optional)
        - notice_text
        - ticker_enabled (boolean)
        - ticker_speed (slow, medium, fast)
        - status (online, offline, unknown)
        - last_heartbeat (datetime, optional)
        - ip_address (optional)
        - api_key_hash
        - created_at, updated_at
      </displays>
      <display_docket_entries>
        - display_id (foreign key)
        - docket_entry_id (foreign key)
        - Composite primary key
      </display_docket_entries>
      <announcements>
        - id (UUID, primary key)
        - text (max 500 chars)
        - priority (integer, lower = higher priority)
        - enabled (boolean)
        - expires_at (datetime, optional)
        - created_at, updated_at, created_by
      </announcements>
      <api_keys>
        - id (UUID, primary key)
        - name
        - key_hash (unique)
        - key_prefix (for identification)
        - permissions (array)
        - display_id (optional, for display-specific keys)
        - expires_at (optional)
        - last_used_at (optional)
        - created_at
      </api_keys>
      <audit_logs>
        - id (UUID, primary key)
        - action (string)
        - entity_type (string)
        - entity_id (optional)
        - user_id (optional)
        - api_key_id (optional)
        - changes (JSON)
        - ip_address (optional)
        - created_at
      </audit_logs>
      <settings>
        - key (string, primary key)
        - value (JSON)
        - updated_at, updated_by
      </settings>
      <calendar_metadata>
        - id (UUID, primary key)
        - judge_name
        - judge_code
        - courtroom (optional)
        - calendar_date (date)
        - generated_at (optional)
        - is_zoom_calendar (boolean)
        - zoom_meeting_id (optional)
        - zoom_passcode (optional)
        - zoom_phone (optional)
        - parameters (optional)
        - source_filename (optional)
        - created_at
        - Unique constraint on (judge_code, calendar_date)
      </calendar_metadata>
    </tables>
  </database_schema>
